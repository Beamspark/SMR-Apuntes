  <div class="feedback">

<details class="quiz-option incorrect">

  <div class="feedback">
    <div class="feedback-title">✗ Incorrecto</div>

<details class="quiz-option incorrect">
  <summary>C) Prohíbe la conexión de sensores IoT para evitar el sobrecalentamiento de la planta.</summary>
  <div class="feedback">
    <div class="feedback-title">✗ Incorrecto</div>

  </div>
